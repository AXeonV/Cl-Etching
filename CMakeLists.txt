cmake_minimum_required(VERSION 3.16)
project(chlorine_etch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(etch src/grid.cpp src/surface.cpp src/transport.cpp src/reaction.cpp src/cell_update.cpp src/simulation.cpp)
target_include_directories(etch PUBLIC src)
add_executable(etch_sim src/main.cpp)
target_link_libraries(etch_sim PRIVATE etch)
add_executable(etch_tests tests/test_core.cpp)
target_link_libraries(etch_tests PRIVATE etch)
add_executable(etch_geometry_tests tests/test_geometry.cpp)
target_link_libraries(etch_geometry_tests PRIVATE etch)
enable_testing()
add_test(NAME core COMMAND etch_tests)
add_test(NAME geometry COMMAND etch_geometry_tests)
